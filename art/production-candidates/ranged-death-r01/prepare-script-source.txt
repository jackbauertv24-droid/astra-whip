 }
 const particles=[...flecks.map(p=>({...p,y:stage===0?p.y-20:p.y})),...pile.map(p=>({...p,pile:true,r:p.r*(stage===0?.65:1)}))];

  if(!t)continue;for(let c=0;c<3;c++)output.data[k+c]=Math.round((col[c]*a+output.data[k+c]*b*(1-a))/t);output.data[k+3]=Math.round(t*255);
 }
 for(let k=0;k<output.data.length;k+=4)if(output.data[k+3]<=4)output.data.fill(0,k,k+4);

}
