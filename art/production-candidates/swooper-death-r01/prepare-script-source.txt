#!/usr/bin/env node
// D23 painted limp collapse, local settling, and soft ash erosion.
import{readPNG,writePNG,blank}from'./lib/png.mjs';
import{meshWarp,clamp}from'./lib/deform.mjs';
import{mkdirSync,copyFileSync,readFileSync,writeFileSync}from'node:fs';
const source='art/production-candidates/swooper-death-collapse-r05/key.png';
if(!readFileSync('art/review-swooper-death-key-r05/review_verdict.md','utf8').trimEnd().endsWith('STATIC KEY REVIEW: PASS'))throw Error('Limp source not approved');
const dir='art/production-candidates/swooper-death-r01';mkdirSync(dir,{recursive:true});
copyFileSync('art/frames/swooper_hurt/f001.png',dir+'/f000.png');
copyFileSync(source,dir+'/f001.png');
const image=readPNG(source),smooth=(a,b,x)=>{const t=clamp((x-a)/(b-a));return t*t*(3-2*t);};
const settled=meshWarp(image,[0,60,80,100,120,140,160,180,192,205,220,240,260,280,300,330,384],[0,45,65,80,95,110,128,140,155,170,185,200,215,230,256],(x,y)=>{
 const wing=(1-smooth(155,185,x)+smooth(230,260,x))*smooth(128,145,y)*(1-smooth(195,215,y));
 const hood=smooth(205,215,x)*(1-smooth(240,255,x))*smooth(65,80,y)*(1-smooth(110,128,y));
 return[x+(x<192?2:-2)*wing,y+5*wing+3*hood];
});
for(let k=0;k<settled.data.length;k+=4)if(settled.data[k+3]<=4)settled.data.fill(0,k,k+4);
writePNG(dir+'/f002.png',settled);
let seed=90433;const random=()=>{seed=(Math.imul(seed,1664525)+1013904223)>>>0;return seed/4294967296;};
const ink=[];for(let y=0;y<256;y++)for(let x=0;x<384;x++)if(settled.data[(y*384+x)*4+3]>180)ink.push([x,y]);
const holes=Array.from({length:24},()=>{const [x,y]=ink[Math.floor(random()*ink.length)];return{x,y,r:2+random()*4};});
const flecks=Array.from({length:14},()=>({x:192+(random()-.5)*100,y:128+(random()-.5)*52,r:1+random()*.9}));
// The largest soft ash cluster stays around the authored body anchor; smaller dust drifts outward.
for(let stage=0;stage<2;stage++){
 const out=blank(384,256),opacity=stage===0?.62:.08;
 for(let y=0;y<256;y++)for(let x=0;x<384;x++){
  const k=(y*384+x)*4;if(!settled.data[k+3])continue;
  let mask=1;for(const h of holes){const d=Math.hypot(x-h.x,y-h.y),r=h.r*(stage===0?1:2.4);mask*=clamp((d-r*.4)/(r*.6));}
  const mix=stage===0?.65:.95;for(let c=0;c<3;c++)out.data[k+c]=Math.round(settled.data[k+c]*(1-mix)+[110,106,97][c]*mix);
  out.data[k+3]=Math.round(settled.data[k+3]*opacity*mask);
 }
 const particles=[{x:192,y:128,r:3},...flecks.map(p=>({...p,x:192+(p.x-192)*(stage===0?.5:1),y:128+(p.y-128)*(stage===0?.5:1)}))];
 for(const p of particles)for(let y=Math.floor(p.y-p.r*2);y<=Math.ceil(p.y+p.r*2);y++)for(let x=Math.floor(p.x-p.r*2);x<=Math.ceil(p.x+p.r*2);x++){
  if(x<0||x>=384||y<0||y>=256)continue;
  const distance=Math.hypot(x-p.x,y-p.y)/p.r;if(distance>=1.6)continue;
  const a=(stage===0?210:185)*clamp(1-distance/1.6)/255,k=(y*384+x)*4,b=out.data[k+3]/255,total=a+b*(1-a);
  for(let c=0;c<3;c++)out.data[k+c]=Math.round(([110,106,97][c]*a+out.data[k+c]*b*(1-a))/total);
  out.data[k+3]=Math.round(total*255);
 }
 for(let k=0;k<out.data.length;k+=4)if(out.data[k+3]<=4)out.data.fill(0,k,k+4);
 writePNG(`${dir}/f00${stage+3}.png`,out);
}
writeFileSync(dir+'/derivation.json',JSON.stringify({source,first_frame:'art/frames/swooper_hurt/f001.png',method:'Approved banking hit frame reused at onset, painted unsupported wing collapse, local5px wing sag/3px hood droop, then soft seeded erosion into Ash Grey dust. Root falling is actual180u/s AI motion. No global scale or per-frame recentering.',holes,flecks},null,2)+'\n');
